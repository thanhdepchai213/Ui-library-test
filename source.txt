		return nil

		if isfile and isfile(file) and delfile then delfile(file) end

	end)

		return getcustomasset(file)
	end)
	return okAsset and asset or nil
end

local function safeJSONDecode(text)
    if type(text) ~= "string" then
        return nil
    end
    local ok, result = pcall(function()
        return HttpService:JSONDecode(text)
    end)
    if ok and type(result) == "table" then
        return result
    end
    return nil
end
local function fileAvailable()
    return type(readfile) == "function"
        and type(writefile) == "function"
        and type(isfile) == "function"
end
local function loadLocal(name, fallback)
    if not fileAvailable() then
        return fallback
    end
    local path = FILE_PREFIX .. name .. ".json"
    local existsOK, exists = pcall(isfile, path)
    if not existsOK or not exists then
        return fallback
    end
    local readOK, raw = pcall(readfile, path)
    if not readOK or type(raw) ~= "string" then
        return fallback
    end
    return safeJSONDecode(raw) or fallback
end
local function saveLocal(name, value)
    if not fileAvailable() then
        return
    end
    local path = FILE_PREFIX .. name .. ".json"
    local ok, raw = pcall(function()
        return HttpService:JSONEncode(value)
    end)
    if ok then
        pcall(writefile, path, raw)
    end
end

local function httpGet(url)
    local ok, result = pcall(function()
        return game:HttpGet(url)
    end)
    if ok and type(result) == "string" then
        return result
    end
    local requestFn =
        (syn and syn.request)
        or (http and http.request)
        or http_request
        or request
    if type(requestFn) == "function" then
        local success, response = pcall(function()
            return requestFn({
                Url = url,
                Method = "GET"
            })
        end)
        if success and response then
            return response.Body or response.body
        end
    end
    return nil
end
local function normalizeSong(item)
    if type(item) ~= "table" then
        return nil
    end
    local id =
        item.id
        or item.ID
        or item.Id
        or item.assetId
        or item.AssetId
    local name =
        item.name
        or item.Name
        or item.title
        or item.Title
    local category =
        item.category
        or item.Category
    local cover =
        item.cover
        or item.coverImage
        or item.image
        or item.thumbnail
        or item.thumbnailId
    if not id then
        return nil
    end
    id = tostring(id)
    if not id:match("^%d+$") then
        return nil
    end
    name = tostring(
        name
        or "Untitled Track"
    )
    category = tostring(
        category
        or "Other"
    )
    if category ~= "Phonk"
        and category ~= "Hindi"
        and category ~= "Bhojpuri"
        and category ~= "English"
        and category ~= "Other" then
        category = "Other"
    end
    return {
        id = id,
        name = name,
        category = category,
        cover = cover
    }
end
local function loadRoster()
    local raw = httpGet(ROSTER_URL)
    if not raw then
        notify(
            "SAIKI MUSIC",
            "Could not fetch the music roster."
        )
        return false
    end
    local data = safeJSONDecode(raw)
    if not data then
        notify(
            "SAIKI MUSIC",
            "Roster JSON could not be decoded."
        )
        return false
    end
    local source = data.songs
    if type(source) ~= "table" then
        source = data
    end
    if type(source) ~= "table" then
        notify(
            "SAIKI MUSIC",
            "Roster has no songs."
        )
        return false
    end
    local songs = {}
    local seen = {}
    for _, item in ipairs(source) do
        local song = normalizeSong(item)
        if song and not seen[song.id] then
            seen[song.id] = true
            table.insert(songs, song)
        end
    end
    State.songs = songs
    return #songs > 0
end

local function socialRequest()
    return (syn and syn.request) or (http and http.request) or http_request or request
end
local function downloadSocialAsset(url, fileName)
    local req = socialRequest()
    if not req or not writefile or not getcustomasset then
        return nil
    end
    local ok, res = pcall(function()
        return req({Url = url, Method = "GET"})
    end)
    if not ok or not res or type(res.Body) ~= "string" or #res.Body == 0 then
        return nil
    end
    local wrote = pcall(function()
        if isfile and isfile(fileName) and delfile then
            delfile(fileName)
        end
        writefile(fileName, res.Body)
    end)
    if not wrote then
        return nil
    end
    local okAsset, asset = pcall(function()
        return getcustomasset(fileName)
    end)
    return okAsset and asset or nil
end

    return button
end

local function copySocialLink(url, label)
    local copied = false
    local fn = setclipboard or toclipboard or set_clipboard
    if type(fn) == "function" then
        copied = pcall(fn, url)
    elseif syn and type(syn.write_clipboard) == "function" then
        copied = pcall(syn.write_clipboard, url)
    end
    if copied then
        notify("SAIKI MUSIC", "Copied to clipboard")
    else
        notify("SAIKI MUSIC", "Could not copy " .. label .. " link")
    end
end

local themeOriginals = {}
local function rememberThemeObject(object)
    if themeOriginals[object] then
        return
    end
    themeOriginals[object] = {}
    if object:IsA("GuiObject") then
        themeOriginals[object].BackgroundColor3 = object.BackgroundColor3
        themeOriginals[object].BackgroundTransparency = object.BackgroundTransparency
    end
    if object:IsA("TextLabel")
        or object:IsA("TextButton")
        or object:IsA("TextBox") then
        themeOriginals[object].TextColor3 = object.TextColor3
    end
    if object:IsA("TextBox") then
        themeOriginals[object].PlaceholderColor3 = object.PlaceholderColor3
    end
    if object:IsA("UIStroke") then
        themeOriginals[object].Color = object.Color
        themeOriginals[object].Transparency = object.Transparency
    end
end
local function themeRememberTree()
    rememberThemeObject(Main)
    for _, object in ipairs(Main:GetDescendants()) do
        rememberThemeObject(object)
    end
end
local function restoreLightTheme()
    for object, original in pairs(themeOriginals) do
        if object and object.Parent then
            if original.BackgroundColor3 then
                object.BackgroundColor3 = original.BackgroundColor3
                object.BackgroundTransparency = original.BackgroundTransparency
            end
            if original.TextColor3 then
                object.TextColor3 = original.TextColor3
            end
            if original.PlaceholderColor3 then
                object.PlaceholderColor3 = original.PlaceholderColor3
            end
            if original.Color then
                object.Color = original.Color
                object.Transparency = original.Transparency
            end
        end
    end
end

local function makeDraggable(guiObject, handle)
    handle = handle or guiObject
    local dragging = false
    local dragStart
    local startPosition
    handle.InputBegan:Connect(function(input)
        if anchored then
            return
        end
        if input.UserInputType ==
            Enum.UserInputType.MouseButton1
            or input.UserInputType ==
            Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition =
                guiObject.Position
            local connection
            connection =
                input.Changed:Connect(function()
                    if input.UserInputState ==
                        Enum.UserInputState.End then
                        dragging = false
                        if connection then
                            connection:Disconnect()
                        end
                    end
                end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then
            return
        end
        if input.UserInputType ~=
            Enum.UserInputType.MouseMovement
            and input.UserInputType ~=
            Enum.UserInputType.Touch then
            return
        end
        local delta =
            input.Position - dragStart
        guiObject.Position =
            UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
    end)
end

local function updateFilter()
    local query =
        string.lower(
            State.search or ""
        )
    State.filtered = {}
    for _, song in ipairs(
        State.songs
    ) do
        local categoryOK = false
        if State.category == "Liked" then
            categoryOK =
                State.liked[song.id] == true
        else
            categoryOK =
                song.category ==
                State.category
        end
        if categoryOK then
            local searchable =
                string.lower(
                    tostring(song.name)
                    .. " "
                    .. tostring(song.id)
                )
            if query == ""
                or string.find(
                    searchable,
                    query,
                    1,
                    true
                ) then
                table.insert(
                    State.filtered,
                    song
                )
            end
        end
    end
end

local function fireRemote(
    remote,
    ...
)
    if not remote then
        return false
    end
    local args = {...}
    local ok = pcall(function()
        if remote:IsA(
            "RemoteEvent"
        ) then
            remote:FireServer(
                unpack(args)
            )
        elseif remote:IsA(
            "RemoteFunction"
        ) then
            remote:InvokeServer(
                unpack(args)
            )
        end
    end)
    return ok
end

local function nextSong()
    if #State.queue > 0 then
        local nextItem =
            table.remove(
                State.queue,
                1
            )
        playSong(
            nextItem
        )
        return
    end
    if #State.songs == 0 then
        return
    end
    if State.shuffle
        and #State.songs > 1 then
        local nextIndex =
            State.index
        repeat
            nextIndex =
                math.random(
                    1,
                    #State.songs
                )
        until nextIndex ~=
            State.index
        State.index =
            nextIndex
    else
        State.index =
            State.index + 1
        if State.index >
            #State.songs then
            State.index = 1
        end
    end
    playSong(
        State.songs[
            State.index
        ]
    )
end

local gradientTime = 0
local coverTime = 0
RunService.RenderStepped:Connect(
    function(dt)
        if State.destroyed then
            return
        end
        gradientTime += dt
        coverTime += dt
        HideButton.Rotation =
            (HideButton.Rotation + (dt * 18)) % 360
        local hue =
            (
                gradientTime
                * 0.035
            ) % 1
        local c1
        local c2
        local c3
        if State.theme == "Dark" then
            c1 = Color3.fromHSV((hue + 0.77) % 1, 0.55, 0.16)
            c2 = Color3.fromHSV((hue + 0.66) % 1, 0.45, 0.11)
            c3 = Color3.fromHSV((hue + 0.90) % 1, 0.50, 0.14)
        else
            c1 = Color3.fromHSV((hue + 0.92) % 1, 0.12, 1)
            c2 = Color3.fromHSV((hue + 0.05) % 1, 0.10, 1)
            c3 = Color3.fromHSV((hue + 0.48) % 1, 0.10, 1)
        end
        Gradient.Color =
            ColorSequence.new({
                ColorSequenceKeypoint.new(
                    0,
                    c1
                ),
                ColorSequenceKeypoint.new(
                    0.5,
                    c2
                ),
                ColorSequenceKeypoint.new(
                    1,
                    c3
                )
            })
        if State.theme == "Dark" then
            MainStroke.Color = Color3.fromHSV((hue + 0.86) % 1, 0.52, 0.70)
            CoverStroke.Color = Color3.fromHSV((hue + 0.76) % 1, 0.48, 0.62)
        else
            MainStroke.Color = Color3.fromHSV((hue + 0.9) % 1, 0.18, 0.95)
            CoverStroke.Color = Color3.fromHSV((hue + 0.82) % 1, 0.20, 0.92)
        end
        if State.playing
            and State.current
            and coverTime >= 0.035 then
            coverTime = 0
            State.coverRotation =
                (
                    State.coverRotation
                    + 2
                ) % 360
            Cover.Rotation =
                State.coverRotation
        end
        if cherryVisible and AmbientLayer.Visible then
            for _, leafData in ipairs(cherryLeaves) do
                local leaf = leafData.object
                leafData.y += leafData.speed * dt
                if leafData.y > 1.12 then
                    leafData.y = cherryRandom:NextNumber(-0.18, -0.05)
                    leafData.x = cherryRandom:NextNumber(-0.05, 1.05)
                    leafData.phase = cherryRandom:NextNumber(0, math.pi * 2)
                    leafData.baseTransparency = State.theme == "Dark"
                        and cherryRandom:NextNumber(0.10, 0.22)
                        or cherryRandom:NextNumber(0.08, 0.22)
                end
                local drift
                if State.theme == "Dark" then
                    drift = math.sin(gradientTime * leafData.swaySpeed + leafData.phase) * leafData.sway * 0.22
                else
                    drift = math.sin(gradientTime * leafData.swaySpeed + leafData.phase) * leafData.sway
                end
                local fadeNearBottom = math.clamp((leafData.y - 0.78) / 0.34, 0, 1)
                leaf.Position = UDim2.new(
                    leafData.x + drift,
                    0,
                    leafData.y,
                    0
                )
                leaf.Rotation += (State.theme == "Dark" and leafData.snowRotationSpeed or leafData.rotationSpeed) * dt
                leaf.ImageTransparency = math.clamp(
                    leafData.baseTransparency + fadeNearBottom * (State.theme == "Dark" and 0.78 or 0.70),
                    0,
                    1
                )
            end
        elseif not cherryVisible then
            cherryFade = math.min(1, cherryFade + dt * 2.4)
            for _, leafData in ipairs(cherryLeaves) do
                leafData.object.ImageTransparency = math.min(1, leafData.object.ImageTransparency + dt * 2.4)
            end
        end
        if State.rgb then
            updateVehicleRGB()
        end
    end
)
