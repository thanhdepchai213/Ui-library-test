-- Trạng thái chung
---------------------------------------------------------------------
local State = {
	Running = true,
	DeliverCFrame = nil,
	WalkSpeed = 16,
	JumpPower = 50,
	GrabRange = 300,
}

local Connections = {}

local function remotes()
	local ok, result = pcall(function()
		return require(ReplicatedStorage.Shared.Remotes)
	end)
	return ok and result or nil
end

local function eggState()
	local ok, result = pcall(function()
		return require(ReplicatedStorage.Client.EggState)
	end)
	return ok and result or nil
end

local function invoke(group, name, ...)
	local R = remotes()
	local args = table.pack(...)
	local ok, result = pcall(function()
		return R[group][name]:InvokeServer(table.unpack(args, 1, args.n))
	end)
	return ok, result
end

local function getRoot()
	local char = LocalPlayer.Character
	return char and char:FindFirstChild("HumanoidRootPart"), char and char:FindFirstChildOfClass("Humanoid")
end

-- Vòng lặp có bật/tắt theo Options
local Options = Fluent.Options

local function startLoop(optionId, interval, fn)
	task.spawn(function()
		while State.Running do
			local opt = Options[optionId]
			if opt and opt.Value then
				local ok, err = pcall(fn)
				if not ok then
					warn("[Thành_Adr Hub] " .. optionId .. ": " .. tostring(err))
				end
			end
			task.wait(interval)
		end
	end)
end

local function eggPosition(record)
	local cf = record.BottomCFrame or record.BoundsCFrame
	return cf and cf.Position
end

local function readEggs()
	local ES = eggState()
	if not ES then
		return {}
	end
	local ok, snapshot = pcall(ES.ReadFieldEggs)
	if ok and type(snapshot) == "table" and type(snapshot.Records) == "table" then
		return snapshot.Records
	end
	return {}
end

-- ESP
local espFolder = Instance.new("Folder")
espFolder.Name = "ThanhAdrEggESP"
espFolder.Parent = workspace
local espItems = {}

local function clearEsp()
	for uid, part in pairs(espItems) do
		part:Destroy()
		espItems[uid] = nil
	end
end

local function updateEsp()
	local seen = {}
	for _, record in ipairs(readEggs()) do
		local uid = tostring(record.Uid)
		local pos = eggPosition(record)
		if pos then
			seen[uid] = true
			local part = espItems[uid]
			if not part then
				part = Instance.new("Part")
				part.Anchored = true
				part.CanCollide = false
				part.CanQuery = false
				part.Transparency = 1
				part.Size = Vector3.new(1, 1, 1)
				part.Parent = espFolder

				local gui = Instance.new("BillboardGui")
				gui.Name = "Tag"
				gui.Size = UDim2.fromOffset(170, 34)
				gui.AlwaysOnTop = true
				gui.StudsOffset = Vector3.new(0, 3, 0)
				gui.Adornee = part
				gui.Parent = part

				local label = Instance.new("TextLabel")
				label.Name = "Label"
				label.Size = UDim2.fromScale(1, 1)
				label.BackgroundTransparency = 1
				label.Font = Enum.Font.GothamBold
				label.TextSize = 13
				label.TextColor3 = Color3.fromRGB(255, 214, 102)
				label.TextStrokeTransparency = 0.4
				label.Parent = gui

				espItems[uid] = part
			end
			part.Position = pos
			local label = part:FindFirstChild("Tag") and part.Tag:FindFirstChild("Label")
			if label then
				local rootPart = getRoot()
				local dist = rootPart and math.floor((rootPart.Position - pos).Magnitude) or 0
				label.Text = string.format("%s [%s] %dm", tostring(record.AreaId or "Egg"), tostring(record.State or "?"), dist)
			end
		end
	end
	for uid, part in pairs(espItems) do
		if not seen[uid] then
			part:Destroy()
			espItems[uid] = nil
		end
	end
end

local function slotKeyFor(record)
	local uid = tostring(record.Uid)
	if record.AreaId == "Forest" or uid:find("^FirstArea") then
		local key = "Forest:" .. tostring(record.NestId or "Slot_001")
		pcall(function()
			local identity = require(ReplicatedStorage.Shared.Util.AreaEggSlotIdentity)
			if identity and record.AreaId and record.NestId then
				key = identity.SlotKey(record.AreaId, record.NestId)
			end
		end)
		return key
	end
	return nil
end

local function nearestEgg(root)
	local best, bestDist
	for _, record in ipairs(readEggs()) do
		local free = (record.State == "Slot" or record.State == "Dropped")
			and (not record.CarrierUserId or record.CarrierUserId == LocalPlayer.UserId)
		local pos = eggPosition(record)
		if free and pos then
			local dist = (pos - root.Position).Magnitude
			if dist <= State.GrabRange and (not bestDist or dist < bestDist) then
				best, bestDist = record, dist
			end
		end
	end
	return best
end

startLoop("AutoGrab", 0.7, function()
	local root, humanoid = getRoot()
	if not root or not humanoid or humanoid.Health <= 0 then
		return
	end

	local record = nearestEgg(root)
	if not record then
		return
	end

	local pos = eggPosition(record)
	root.CFrame = CFrame.new(pos.X, pos.Y + 2.5, pos.Z)
	root.AssemblyLinearVelocity = Vector3.zero
	task.wait(0.15)

	local params = { Uid = record.Uid }
	local slotKey = slotKeyFor(record)
	if slotKey then
		params.FirstAreaSlotKey = slotKey
	end
	invoke("EggWorld", "AskFieldEggCarry", params)

	local ES = eggState()
	local carried = false
	if ES then
		local ok, current = pcall(ES.ReadFieldEgg, record.Uid)
		carried = ok and type(current) == "table" and current.State == "Carried" and current.CarrierUserId == LocalPlayer.UserId
	end

	if carried and State.DeliverCFrame then
		task.wait(0.2)
		local rootNow = getRoot()
		if rootNow then
			rootNow.CFrame = State.DeliverCFrame
			rootNow.AssemblyLinearVelocity = Vector3.zero
		end
		task.wait(1)
	end
end)
