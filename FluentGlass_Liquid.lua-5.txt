-- ========== Tiện ích ==========
local GlassItems = {}   -- mọi khung kính (để chỉnh độ kính / theme)
local AccentItems = {}  -- {object, property} dùng màu nhấn
local Irid = {}         -- gradient viền cầu vồng
local iridConn

local function new(class, props, children)
    local o = Instance.new(class)
    if class == "TextLabel" or class == "TextButton" or class == "TextBox" then
        o.TextStrokeColor3 = Color3.fromRGB(10, 20, 40)
        o.TextStrokeTransparency = 0.9
    end
    for k, v in pairs(props or {}) do o[k] = v end
    for _, c in ipairs(children or {}) do c.Parent = o end
    return o
end

local function tween(o, t, props, style, dir)
    local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function kp(t, v) return NumberSequenceKeypoint.new(t, v) end
local function corner(r) return new("UICorner", { CornerRadius = UDim.new(0, r) }) end
local function pad(l, t, r, b)
    return new("UIPadding", {
        PaddingLeft = UDim.new(0, l), PaddingTop = UDim.new(0, t),
        PaddingRight = UDim.new(0, r or l), PaddingBottom = UDim.new(0, b or t),
    })
end

local function accentize(obj, prop)
    table.insert(AccentItems, { obj, prop })
    obj[prop] = Library.Accent
end

local function isPress(i)
    return i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch
end
local function isMove(i)
    return i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch
end
-- Vị trí trong hệ toạ độ ScreenGui (IgnoreGuiInset = true) -> cộng thêm inset
local function screenPos(i)
    return Vector2.new(i.Position.X, i.Position.Y) + GuiService:GetGuiInset()
end

local function ensureIridAnim()
    if iridConn then return end
    local t0 = os.clock()
    iridConn = RunService.Heartbeat:Connect(function()
        local rot = (225 + (os.clock() - t0) * 45) % 360
        for i = #Irid, 1, -1 do
            local g = Irid[i]
            if g.Parent then g.Rotation = rot else table.remove(Irid, i) end
        end
    end)
end

-- Viền kính. uniform=true: viền đều kiểu "outline" (hàng, nút); false: sáng 2 góc (cửa sổ)
local function glassStroke(parent, thickness, strength, uniform)
    local s = new("UIStroke", {
        Thickness = thickness or 1.5, Color = WHITE, Transparency = strength or 0.3,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
    local seq = uniform and NumberSequence.new({ kp(0, 0), kp(0.5, 0.4), kp(1, 0) })
        or NumberSequence.new({ kp(0, 0), kp(0.3, 0.6), kp(0.5, 0.92), kp(0.75, 0.6), kp(1, 0.05) })
    new("UIGradient", { Rotation = 45, Transparency = seq }).Parent = s
    s.Parent = parent
    return s
end

local function glassShine(parent, radius)
    local f = new("Frame", {
        Name = "Shine", BackgroundColor3 = WHITE, BackgroundTransparency = 0.8,
        Size = UDim2.fromScale(1, 1), BorderSizePixel = 0, ZIndex = parent.ZIndex, Active = false,
    }, { corner(radius) })
    new("UIGradient", {
        Rotation = 90,
        Transparency = NumberSequence.new({ kp(0, 0.2), kp(0.25, 0.75), kp(0.7, 1), kp(1, 0.7) }),
    }).Parent = f
    f.Parent = parent
    return f
end

-- o.light = true: chỉ nền + viền + bóng (nhẹ, dùng cho hàng/nút nhỏ)
-- o.themed = true: màu nền theo theme
-- o.uniform = true: viền đều
local function glass(parent, props, radius, transparency, o)
    o = o or {}
    local r = radius or 12
    local f = new("Frame", { BackgroundColor3 = WHITE, BackgroundTransparency = transparency or 0.5, BorderSizePixel = 0 })
    for k, v in pairs(props or {}) do f[k] = v end
    corner(r).Parent = f
    glassStroke(f, o.uniform and 1.3 or 1.5, o.uniform and 0.45 or 0.3, o.uniform)
    glassShine(f, r)
    if not o.light then
        -- đáy tối nhẹ: tách lớp kính
        local depth = new("Frame", {
            Name = "Depth", BackgroundColor3 = Color3.new(0, 0, 0), BorderSizePixel = 0,
            Size = UDim2.fromScale(1, 1), Active = false,
        }, { corner(r) })
        new("UIGradient", { Rotation = 90, Transparency = NumberSequence.new({ kp(0, 1), kp(0.6, 1), kp(1, 0.82) }) }).Parent = depth
        depth.Parent = f
        -- viền khúc xạ trong, màu cầu vồng xoay
        if r >= 16 then
            local rim = new("Frame", {
                Name = "Rim", BackgroundTransparency = 1, Position = UDim2.fromOffset(3, 3),
                Size = UDim2.new(1, -6, 1, -6), Active = false,
            }, { corner(r - 3) })
            local st = new("UIStroke", { Thickness = 1.3, Color = WHITE, Transparency = 0.4, ApplyStrokeMode = Enum.ApplyStrokeMode.Border })
            local ig = new("UIGradient", {
                Rotation = 225,
                Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(140, 230, 255)),
                    ColorSequenceKeypoint.new(0.35, WHITE),
                    ColorSequenceKeypoint.new(0.65, Color3.fromRGB(255, 160, 235)),
                    ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 235, 150)),
                }),
                Transparency = NumberSequence.new({ kp(0, 0.1), kp(0.35, 0.85), kp(0.65, 0.85), kp(1, 0.1) }),
            })
            ig.Parent = st
            table.insert(Irid, ig)
            ensureIridAnim()
            st.Parent = rim
            rim.Parent = f
        end
    end
    if not o.light and r >= 20 then
        -- tán sắc: hai viền đỏ/xanh lệch nhau 1px như mép thấu kính thật
        for _, c in ipairs({ { Color3.fromRGB(255, 90, 130), 1 }, { Color3.fromRGB(80, 210, 255), -1 } }) do
            local ch = new("Frame", {
                Name = "Chroma", BackgroundTransparency = 1, Position = UDim2.fromOffset(c[2], 0),
                Size = UDim2.fromScale(1, 1), Active = false,
            }, { corner(r) })
            new("UIStroke", { Thickness = 1, Color = c[1], Transparency = 0.8, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }).Parent = ch
            ch.Parent = f
        end
    end
    f:SetAttribute("GT", f.BackgroundTransparency)
    table.insert(GlassItems, {
        f = f, base = f.BackgroundColor3, themed = o.themed,
        clear = math.min(0.98, f.BackgroundTransparency + 0.3), tinted = math.max(0.2, f.BackgroundTransparency - 0.3),
    })
    f.Parent = parent
    return f
end

local function applyGlass(animate)
    local level = Library.GlassLevel
    local alive = {}
    for _, it in ipairs(GlassItems) do
        if it.f and it.f.Parent then
            table.insert(alive, it)
            local t = it.clear + (it.tinted - it.clear) * level
            local base = it.themed and Library.Tint or it.base
            local col
            if level < 0.5 then col = base:Lerp(WHITE, (0.5 - level) * 0.6)
            else col = base:Lerp(Color3.fromRGB(24, 28, 38), (level - 0.5) * 1.4) end
            it.f:SetAttribute("GT", t)
            if animate then tween(it.f, 0.25, { BackgroundTransparency = t, BackgroundColor3 = col })
            else it.f.BackgroundTransparency = t; it.f.BackgroundColor3 = col end
        end
    end
    GlassItems = alive
    for _, w in ipairs(Library.Windows) do
        if w.Blur and w.Visible then tween(w.Blur, 0.3, { Size = 14 + level * 14 }) end
    end
end

function Library:SetGlass(level)
    Library.GlassLevel = math.clamp(level or 0.5, 0, 1)
    applyGlass(true)
end

function Library:SetEffects(on)
    Library.Effects = on and true or false
    for _, w in ipairs(Library.Windows) do
        if w.Fx and w.Fx.Parent then w.Fx.Visible = Library.Effects end
    end
end

function Library:SetTheme(name)
    local th = Library.Themes[name]
    if not th then return end
    Library.ThemeName, Library.Tint, Library.Accent = name, th.Tint, th.Accent
    local alive = {}
    for _, e in ipairs(AccentItems) do
        if e[1] and e[1].Parent then table.insert(alive, e); tween(e[1], 0.3, { [e[2]] = th.Accent }) end
    end
    AccentItems = alive
    applyGlass(true)
end

local function gui_parent()
    local ok, h = pcall(function() return gethui and gethui() end)
    if ok and h then return h end
    local ok2 = pcall(function() return game:GetService("CoreGui"):GetChildren() end)
    if ok2 then return game:GetService("CoreGui") end
    return Players.LocalPlayer:WaitForChild("PlayerGui")
end

local notifyGui, notifyHolder
function Library:Notify(title, content, duration)
    if not notifyGui or not notifyGui.Parent then
        notifyGui = new("ScreenGui", { Name = "FluentGlassNotify", ResetOnSpawn = false, IgnoreGuiInset = true, DisplayOrder = 999 })
        notifyGui.Parent = gui_parent()
        notifyHolder = new("Frame", {
            BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0),
            Position = UDim2.new(1, -12, 0, 12), Size = UDim2.fromOffset(270, 400),
        }, { new("UIListLayout", { Padding = UDim.new(0, 8), HorizontalAlignment = Enum.HorizontalAlignment.Right, SortOrder = Enum.SortOrder.LayoutOrder }) })
        notifyHolder.Parent = notifyGui
    end
    local card = glass(notifyHolder, { Size = UDim2.fromOffset(270, 0), AutomaticSize = Enum.AutomaticSize.Y }, 20, 0.4, { themed = true })
    card.Position = UDim2.fromOffset(320, 0)
    pad(14, 10).Parent = card
    local list = new("UIListLayout", { Padding = UDim.new(0, 2), SortOrder = Enum.SortOrder.LayoutOrder })
    -- UIListLayout chỉ áp cho 2 nhãn: đặt trong khung chứa riêng
    local body = new("Frame", { BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, ZIndex = 3, Parent = card }, { list })
    new("TextLabel", {
        BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 18), Font = Enum.Font.GothamBold, TextSize = 14, LayoutOrder = 1,
        TextColor3 = TEXT, TextXAlignment = Enum.TextXAlignment.Left, Text = tostring(title), ZIndex = 3, Parent = body,
    })
    new("TextLabel", {
        BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, LayoutOrder = 2,
        Font = Enum.Font.Gotham, TextSize = 13, TextWrapped = true, TextColor3 = SUB,
        TextXAlignment = Enum.TextXAlignment.Left, Text = tostring(content or ""), ZIndex = 3, Parent = body,
    })
    tween(card, 0.7, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Elastic)
    local function dismiss()
        if card.Parent then
            tween(card, 0.3, { Position = UDim2.fromOffset(320, 0) })
            task.wait(0.32)
            card:Destroy()
        end
    end
    if duration == 0 then
        local x = new("TextButton", {
            BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, 0, 0, -4), Size = UDim2.fromOffset(22, 22),
            Text = "✕", Font = Enum.Font.GothamBold, TextSize = 13, TextColor3 = SUB, ZIndex = 6, Parent = card,
        })
        x.MouseButton1Click:Connect(function() task.spawn(dismiss) end)
    else
        task.delay(duration or 4, dismiss)
    end
end

-- ========== Panel dropdown nổi ==========
local activePanel
local function closePanel()
    if activePanel then
        local p = activePanel
        activePanel = nil
        p()
    end
end

local function openPanel(screen, anchor, options, selected, onPick)
    closePanel()
    local catcher = new("TextButton", {
        BackgroundTransparency = 1, Text = "", AutoButtonColor = false, Size = UDim2.fromScale(1, 1), ZIndex = 60, Parent = screen,
    })
    local ap, as, vp = anchor.AbsolutePosition, anchor.AbsoluteSize, screen.AbsoluteSize
    local pw = math.max(as.X, 220)
    local ph = math.min(64 + #options * 40, 290)
    local x = math.clamp(ap.X + as.X - pw, 8, math.max(8, vp.X - pw - 8))
    local y = ap.Y + as.Y + 8
    if y + ph > vp.Y - 8 then y = math.max(8, ap.Y - ph - 8) end

    local panel = glass(screen, {
        Size = UDim2.fromOffset(as.X, as.Y), Position = UDim2.fromOffset(ap.X, ap.Y), ZIndex = 61, ClipsDescendants = true,
    }, 22, 0.38, { themed = true })
    applyGlass(false)

    local sbox = new("Frame", {
        BackgroundColor3 = WHITE, BackgroundTransparency = 0.9, Position = UDim2.fromOffset(10, 10), Size = UDim2.new(1, -20, 0, 34), ZIndex = 63, Parent = panel,
    }, { corner(17) })
    glassStroke(sbox, 1.2, 0.45, true)
    new("TextLabel", { BackgroundTransparency = 1, Position = UDim2.fromOffset(12, 0), Size = UDim2.fromOffset(18, 34), Text = "⌕", Font = Enum.Font.GothamBold, TextSize = 17, TextColor3 = SUB, ZIndex = 64, Parent = sbox })
    local search = new("TextBox", {
        BackgroundTransparency = 1, Position = UDim2.fromOffset(34, 0), Size = UDim2.new(1, -44, 1, 0), ClearTextOnFocus = false, Text = "",
        PlaceholderText = "Search...", PlaceholderColor3 = SUB, Font = Enum.Font.Gotham, TextSize = 13, TextColor3 = TEXT,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 64, Parent = sbox,
    })

    local list = new("ScrollingFrame", {
        BackgroundTransparency = 1, Position = UDim2.fromOffset(6, 52), Size = UDim2.new(1, -12, 1, -58), BorderSizePixel = 0,
        CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollBarThickness = 3, ScrollBarImageTransparency = 0.4, ZIndex = 63, Parent = panel,
    }, { new("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder }) })

    local function build(filter)
        for _, c in ipairs(list:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
        for idx, opt in ipairs(options) do
            local name = tostring(opt)
            if filter == "" or string.find(string.lower(name), filter, 1, true) then
                local on = (opt == selected)
                local b = new("TextButton", {
                    BackgroundColor3 = WHITE, BackgroundTransparency = on and 0.82 or 1, Size = UDim2.new(1, -4, 0, 36), LayoutOrder = idx,
                    AutoButtonColor = false, Text = name, Font = Enum.Font.Gotham, TextSize = 14, TextColor3 = TEXT,
                    TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 64, Parent = list,
                }, { corner(18), pad(16, 0, 30, 0) })
                if on then
                    glassStroke(b, 1.3, 0.35, true)
                    local bar = new("Frame", { BorderSizePixel = 0, Position = UDim2.new(0, 3, 0.5, -9), Size = UDim2.fromOffset(3, 18), ZIndex = 65, Parent = b }, { corner(2) })
                    accentize(bar, "BackgroundColor3")
                    new("TextLabel", { BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -4, 0.5, 0), Size = UDim2.fromOffset(16, 16), Text = "●", TextSize = 10, Font = Enum.Font.GothamBold, TextColor3 = TEXT, ZIndex = 65, Parent = b })
                end
                b.MouseEnter:Connect(function() if not on then tween(b, 0.15, { BackgroundTransparency = 0.9 }) end end)
                b.MouseLeave:Connect(function() if not on then tween(b, 0.2, { BackgroundTransparency = 1 }) end end)
                b.MouseButton1Click:Connect(function()
                    selected = opt
                    onPick(opt)
                    closePanel()
                end)
            end
        end
    end
    build("")
    search:GetPropertyChangedSignal("Text"):Connect(function() build(string.lower(search.Text)) end)

    tween(panel, 0.8, { Size = UDim2.fromOffset(pw, ph), Position = UDim2.fromOffset(x, y) }, Enum.EasingStyle.Elastic)

    activePanel = function()
        catcher:Destroy()
        tween(panel, 0.25, { Size = UDim2.fromOffset(as.X, as.Y), Position = UDim2.fromOffset(ap.X, ap.Y), BackgroundTransparency = 1 })
        task.delay(0.27, function() panel:Destroy() end)
    end
    catcher.MouseButton1Click:Connect(closePanel)
end

-- ========== Cửa sổ ==========
function Library:CreateWindow(cfg)
    cfg = cfg or {}
    if cfg.Theme then Library:SetTheme(cfg.Theme) end
    local Window = { Tabs = {}, Visible = true, Maximized = false }

    local screen = new("ScreenGui", { Name = "FluentGlassLiquid", ResetOnSpawn = false, IgnoreGuiInset = true, DisplayOrder = 100, ZIndexBehavior = Enum.ZIndexBehavior.Sibling })
    screen.Parent = gui_parent()
    Window.Gui = screen

    local blur
    if cfg.Blur ~= false then
        blur = new("BlurEffect", { Name = "FluentGlassBlur", Size = 0 })
        blur.Parent = Lighting
        Window.Blur = blur
    end

    local cam = workspace.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(900, 600)
    local w = math.min(cfg.Width or 660, vp.X - 24)
    local h = math.min(cfg.Height or 410, vp.Y - 48)
    local compact = w < 540
    local sideW = compact and 56 or 190
    local curW, curH = w, h

    local main = glass(screen, {
        Name = "Main", AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromOffset(w, h),
    }, 28, 0.42, { themed = true })
    applyGlass(false)
    Window.Main = main

    main.Size = UDim2.fromOffset(w * 0.9, h * 0.9)
    tween(main, 0.9, { Size = UDim2.fromOffset(w, h) }, Enum.EasingStyle.Elastic)
    if blur then tween(blur, 0.6, { Size = 14 + Library.GlassLevel * 14 }) end

    -- Vạch sáng mép trên + đèn rọi bám theo ngón tay
    local topLine = new("Frame", { BackgroundColor3 = WHITE, BorderSizePixel = 0, Active = false, ZIndex = 2, Position = UDim2.new(0, 40, 0, 2), Size = UDim2.new(1, -80, 0, 1), Parent = main })
    new("UIGradient", { Transparency = NumberSequence.new({ kp(0, 1), kp(0.5, 0.15), kp(1, 1) }) }).Parent = topLine

    local glowClip = new("Frame", { BackgroundTransparency = 1, ClipsDescendants = true, Active = false, ZIndex = 2, Position = UDim2.fromOffset(14, 14), Size = UDim2.new(1, -28, 1, -28), Parent = main })
    -- Lớp "chất lỏng": cực quang trôi chậm + bong bóng nổi (tắt bằng Library:SetEffects(false))
    local fx = new("Frame", { Name = "Fx", BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1), Active = false, ZIndex = 2, Visible = Library.Effects, Parent = glowClip })
    Window.Fx = fx
    local blobs, bubbles = {}, {}
    for k = 1, 3 do
        local root = new("Frame", { BackgroundTransparency = 1, Size = UDim2.fromOffset(0, 0), Active = false, ZIndex = 2, Parent = fx })
        for _, d in ipairs({ { 300, 0.95 }, { 190, 0.93 }, { 100, 0.91 } }) do
            local c = new("Frame", {
                AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(d[1], d[1]), BorderSizePixel = 0,
                BackgroundTransparency = d[2], Active = false, ZIndex = 2, Parent = root,
            }, { new("UICorner", { CornerRadius = UDim.new(1, 0) }) })
            accentize(c, "BackgroundColor3")
        end
        blobs[k] = root
    end
    for k = 1, 8 do
        local sz = 12 + (k * 7) % 28
        local b = new("Frame", {
            BackgroundColor3 = WHITE, BackgroundTransparency = 0.92, Size = UDim2.fromOffset(sz, sz), BorderSizePixel = 0,
            Active = false, ZIndex = 2, Parent = fx,
        }, { new("UICorner", { CornerRadius = UDim.new(1, 0) }) })
        new("UIStroke", { Thickness = 1, Color = WHITE, Transparency = 0.5, Parent = b })
        bubbles[k] = { f = b, x = (k * 0.137) % 1, y = (k * 0.31) % 1, sp = 0.02 + 0.012 * (k % 4), ph = k * 1.7 }
    end
    local fxConn = RunService.Heartbeat:Connect(function(dt)
        if not Library.Effects or not main.Visible then return end
        local t, sz = os.clock(), glowClip.AbsoluteSize
        for k, root in ipairs(blobs) do
            root.Position = UDim2.fromOffset(sz.X * (0.5 + 0.42 * math.sin(t * 0.23 * k + k)), sz.Y * (0.5 + 0.42 * math.cos(t * 0.19 * k + k * 2)))
        end
        for _, bb in ipairs(bubbles) do
            bb.y = bb.y - bb.sp * dt
            if bb.y < -0.1 then bb.y = 1.1 end
            bb.f.Position = UDim2.fromOffset(sz.X * (bb.x + 0.03 * math.sin(t + bb.ph)), sz.Y * bb.y)
        end
    end)
    local glowRoot = new("Frame", { BackgroundTransparency = 1, Size = UDim2.fromOffset(0, 0), Active = false, ZIndex = 2, Parent = glowClip })

            end
            set(value, true)
            function api:Set(v) set(v) end
            function api:Get() return value end
            local dragging = false
            local function fromX(x) set(lo + math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1) * (hi - lo)) end
            bar.InputBegan:Connect(function(i)
                if isPress(i) then
                    dragging = true; page.ScrollingEnabled = false; fromX(i.Position.X)
                    tween(knob, 0.35, { Size = UDim2.fromOffset(28, 22) }, Enum.EasingStyle.Back)
                end
            end)
            UIS.InputChanged:Connect(function(i) if dragging and isMove(i) then fromX(i.Position.X) end end)
            UIS.InputEnded:Connect(function(i)
                if isPress(i) and dragging then
                    dragging = false; page.ScrollingEnabled = true
                    tween(knob, 0.6, { Size = UDim2.fromOffset(16, 16) }, Enum.EasingStyle.Elastic)
                end
            end)
            return api
        end

        function Tab:AddDropdown(a, options, default, cb, desc)
            local o = type(a) == "table" and a or { Title = a, Options = options, Default = default, Callback = cb, Description = desc }
            local opts = o.Options or {}
            local selected = o.Default
            local r, scale = card(o.Description and 60 or 48, o.Title, o.Description, 190)
            local pill = glass(r, { AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -14, 0.5, 0), Size = UDim2.fromOffset(176, 32), ZIndex = 7 }, 16, 0.9, { light = true, uniform = true })
            local cur = new("TextLabel", {
                BackgroundTransparency = 1, Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -40, 1, 0), Font = Enum.Font.Gotham, TextSize = 13, TextColor3 = TEXT,
                TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Text = tostring(selected or "Chọn..."), ZIndex = 8, Parent = pill,
            })
            new("TextLabel", { BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -10, 0, 0), Size = UDim2.fromOffset(16, 32), Text = "›", Font = Enum.Font.GothamBold, TextSize = 18, TextColor3 = TEXT, ZIndex = 8, Parent = pill })
            local api = {}
            function api:Set(v, silent) selected = v; cur.Text = tostring(v); if o.Callback and not silent then task.spawn(o.Callback, v) end end
            function api:Get() return selected end
            function api:SetOptions(list) opts = list end
            clickable(r, scale, function()
                openPanel(screen, pill, opts, selected, function(v) api:Set(v) end)
            end)
            return api
        end

        function Tab:AddInput(a, placeholder, cb, desc)
            local o = type(a) == "table" and a or { Title = a, Placeholder = placeholder, Callback = cb, Description = desc }
            local r = card(o.Description and 60 or 48, o.Title, o.Description, 200)
            local pill = glass(r, { AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -14, 0.5, 0), Size = UDim2.fromOffset(186, 32), ZIndex = 7 }, 16, 0.9, { light = true, uniform = true })
            local box = new("TextBox", {
                BackgroundTransparency = 1, Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -28, 1, 0), ClearTextOnFocus = false, Text = "",
                PlaceholderText = o.Placeholder or "", PlaceholderColor3 = SUB, Font = Enum.Font.Gotham, TextSize = 13, TextColor3 = TEXT,
                TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 8, Parent = pill,
            })
            local st = pill:FindFirstChildOfClass("UIStroke")
            box.Focused:Connect(function() tween(st, 0.2, { Transparency = 0, Thickness = 2 }) end)
            box.FocusLost:Connect(function(enter)
                tween(st, 0.2, { Transparency = 0.45, Thickness = 1.3 })
                if o.Callback then task.spawn(o.Callback, box.Text, enter) end
            end)
            return { Set = function(_, t) box.Text = tostring(t) end, Get = function() return box.Text end }
        end

        return Tab
    end

    table.insert(Library.Windows, Window)
    return Window
end
