local function PlayerClick()
  local Char = Player.Character
  if Char then
    if Settings.AutoClick then
      VirtualUser:CaptureController()
      VirtualUser:Button1Down(Vector2.new(1e4, 1e4))
    end
    if Settings.AutoHaki and Char:FindFirstChild("AuraColor_Folder") and Funcs:AbilityUnlocked("Aura") then
      if #Char.AuraColor_Folder:GetChildren() < 1 then
        OtherEvent.MainEvents.Ability:InvokeServer("Aura")
      end
    end
  end
end

local function IsAlive(Char)
  local Hum = Char and Char:FindFirstChild("Humanoid")
  return Hum and Hum.Health > 0
end

local function GetNextEnemie(EnemieName)
  for _,v in ipairs(Monsters:GetChildren()) do
    if (not EnemieName or v.Name == EnemieName) and IsAlive(v) then
      return v
    end
  end
  return false
end

local function GoTo(CFrame, Move)
  local Char = Player.Character
  if IsAlive(Char) then
    return Move and ( Char:MoveTo(CFrame.p) or true ) or Char:SetPrimaryPartCFrame(CFrame)
  end
end

local function EquipWeapon()
  local Backpack, Char = Player:FindFirstChild("Backpack"), Player.Character
  if IsAlive(Char) and Backpack then
    for _,v in ipairs(Backpack:GetChildren()) do
      if v:IsA("Tool") and v.ToolTip:find(Settings.ToolFarm) then
        Char.Humanoid:EquipTool(v)
      end
    end
  end
end

local function BringMobsTo(_Enemie, CFrame, SBring)
  for _,v in ipairs(Monsters:GetChildren()) do
    if (SBring or v.Name == _Enemie) and IsAlive(v) then
      local PP, Hum = v.PrimaryPart, v.Humanoid
      if PP and (PP.Position - CFrame.p).Magnitude < 500 then
        Hum.WalkSpeed = 0
        Hum:ChangeState(14)
        PP.CFrame = CFrame
        PP.CanCollide = false
        PP.Transparency = Settings.ViewHitbox and 0.8 or 1
        PP.Size = Vector3.new(50, 50, 50)
      end
    end
  end
  return pcall(sethiddenproperty, Player, "SimulationRadius", _huge)
end

local function KillMonster(_Enemie, SBring)
  local Enemy = typeof(_Enemie) == "Instance" and _Enemie or GetNextEnemie(_Enemie)
  if IsAlive(Enemy) and Enemy.PrimaryPart then
    GoTo(Enemy.PrimaryPart.CFrame * Settings.FarmCFrame)EquipWeapon()
    if not Enemy:FindFirstChild("Reverse_Mark") then PlayerClick() end
    if Settings.BringMobs then BringMobsTo(_Enemie, Enemy.PrimaryPart.CFrame, SBring) end
    return true
  end
end

local function TakeQuest(QuestName, CFrame, Wait)
  local QuestGiver = Quests_Npc:FindFirstChild(QuestName)
  if QuestGiver and Player:DistanceFromCharacter(QuestGiver.WorldPivot.p) < 5 then
    return fireproximityprompt(QuestGiver.Block.QuestPrompt), _wait(Wait or 0.1)
  end
  GoTo(CFrame or QuestLocation[QuestName].CFrame)
end

local function ClearQuests(Ignore)
  for _,v in ipairs(QuestFolder:GetChildren()) do
    if v.QuestGiver.Value ~= Ignore and v.Target.Value ~= "None" then
      OtherEvent.QuestEvents.Quest:FireServer("Abandon_Quest", { QuestSlot = v.Name })
    end
  end
end

local function GetRaidEnemies()
  for _,v in ipairs(Monsters:GetChildren()) do
    if v:GetAttribute("Raid_Enemy") and IsAlive(v) then
      return v
    end
  end
end

local function GetRaidMap()
  for _,v in ipairs(Raids:GetChildren()) do
    if v.Joiners:FindFirstChild(Player.Name) then
      return v
    end
  end
end

local function VerifyQuest(QName)
  local Quest = Funcs:CheckQuest()
  return Quest and Quest.QuestGiver.Value == QName
end

		local function DummyControl()
			local c = {}
			function c:Refresh(_) end
			function c:RefreshDesc(_) end
			function c:Update(_) end
			return c
		end

		local function WrapControl(control)
			if not control then return DummyControl() end
			local proxy = { _control = control }
			function proxy:Refresh(value)
				if self._control.Refresh then pcall(self._control.Refresh, self._control, value) end
			end
			function proxy:RefreshDesc(value)
				if self._control.RefreshDesc then pcall(self._control.RefreshDesc, self._control, value)
				elseif self._control.SetDesc then pcall(self._control.SetDesc, self._control, value) end
			end
			function proxy:Update(value)
				if self._control.Update then pcall(self._control.Update, self._control, value) end
			end
			return proxy
		end

		local function WrapTab(tab)
			local proxy = { _tab = tab }

			function proxy:addSection(title)
				if title and title ~= "" and self._tab.AddSection then
					pcall(self._tab.AddSection, self._tab, title)
				end
				return self
			end

			function proxy:addMenu(title)
				-- RedzLib has no separate menu container in this API; keep the
				-- hierarchy as section headers while placing controls on the same tab.
				if title and title ~= "" and self._tab.AddSection then
					pcall(self._tab.AddSection, self._tab, title)
				end
				return self
			end

			function proxy:addLabel(title, desc)
				if self._tab.AddParagraph then
					return WrapControl(self._tab:AddParagraph(title, desc or ""))
				end
				return DummyControl()
			end

			function proxy:addButton(title, callback)
				if self._tab.AddButton then
					return WrapControl(self._tab:AddButton({
						Name = title,
						Description = "",
						Callback = callback or function() end
					}))
				end
				return DummyControl()
			end

			function proxy:addButtonGrid(title, buttons)
				if type(buttons) == "table" then
					for key, item in pairs(buttons) do
						if type(item) == "function" then
							self:addButton(tostring(key), item)
						elseif type(item) == "table" then
							self:addButton(item.Name or item.Label or item.Text or tostring(key), item.Callback or item.Function or function() end)
						end
					end
				end
				return self
			end

			function proxy:addToggle(name, default, callback, locked, description, save)
				if self._tab.AddToggle then
					return WrapControl(self._tab:AddToggle({
						Name = name,
						Description = description or "",
						Default = default == true,
						Callback = callback or function() end
					}))
				end
				return DummyControl()
			end

			function proxy:addDropdown(name, default, options, callback, locked, arg8, save)
				if self._tab.AddDropdown then
					local opts = options or {}
					local redzDefault = default
					-- Numeric option index converted for RedzLib
					-- the selected option value (string/table), not a number.
					if type(redzDefault) == "number" then
						redzDefault = opts[redzDefault]
					end
					return WrapControl(self._tab:AddDropdown({
						Name = name,
						Options = opts,
						Default = redzDefault,
						Callback = callback or function() end
					}))
				end
				return DummyControl()
			end

			function proxy:addSlider(name, min, max, default, callback, locked, step, save)
				if self._tab.AddSlider then
					local increment = tonumber(step) or 1
					if increment <= 0 then increment = 1 end
					return WrapControl(self._tab:AddSlider({
						Name = name,
						Min = min,
						Max = max,
						Default = default,
						Increment = increment,
						Callback = callback or function() end
					}))
				end
				return DummyControl()
			end

			function proxy:addTextbox(name, callback, confirmText, save)
				if self._tab.AddInput then
					return WrapControl(self._tab:AddInput({
						Name = name,
						Placeholder = "",
						Callback = callback or function() end
					}))
				elseif self._tab.AddTextbox then
					return WrapControl(self._tab:AddTextbox({
						Name = name,
						Default = "",
						TextDisappear = false,
						Callback = callback or function() end
					}))
				end
				return DummyControl()
			end

			return proxy
		end

local _Stats = Tabs.Stats do
  local StatsName = {
    Power = "MemePowerLevel", Health = "DefenseLevel",
    Weapon = "SwordLevel", Melee = "MeleeLevel"
  }
  local SelectedStats = {}

  _Stats:addSlider("Số điểm mỗi lần cộng", 1, 100, Settings.AutoStats_Points, function(Value)
    Settings.AutoStats_Points = Value
  end, false, 1)
  _Stats:addToggle("Tự động cộng chỉ số", false, function(Value)
    _env.AutoStats = Value
    if not Value then return end
    task.spawn(function()
      local _Points = PlayerData.SkillPoint
      while _env.AutoStats do _wait(0.5)
        for Name, On in pairs(SelectedStats) do
          local Pts, Stat = _Points.Value, PlayerData[StatsName[Name]]
          if On and Pts > 0 and Stat.Value < MSetting.Setting.MaxLevel then
            OtherEvent.MainEvents.StatsFunction:InvokeServer({
              ["Target"] = StatsName[Name],
              ["Action"] = "UpgradeStats",
              ["Amount"] = math.clamp(Settings.AutoStats_Points, 0, MSetting.Setting.MaxLevel - Stat.Value)
            })
          end
        end
      end
    end)
  end)
  _Stats:addSection("Chọn chỉ số")
  for _,Name in ipairs({"Power", "Health", "Weapon", "Melee"}) do
    _Stats:addToggle(Name, false, function(Value) SelectedStats[Name] = Value end)
  end
end
