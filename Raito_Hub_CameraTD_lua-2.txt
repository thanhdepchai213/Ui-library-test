-- ========== Tiện ích ==========
local GlassItems = {}   -- mọi khung kính (để chỉnh độ kính / theme)
local AccentItems = {}  -- {object, property} dùng màu nhấn
local Irid = {}         -- gradient viền cầu vồng
local iridConn

local function new(class, props, children)
    local o = Instance.new(class)
    if class == "TextLabel" or class == "TextButton" or class == "TextBox" then
        o.TextStrokeColor3 = Color3.fromRGB(10, 20, 40)
        o.TextStrokeTransparency = 0.9
    end
    for k, v in pairs(props or {}) do o[k] = v end
    for _, c in ipairs(children or {}) do c.Parent = o end
    return o
end

local function tween(o, t, props, style, dir)
    local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function kp(t, v) return NumberSequenceKeypoint.new(t, v) end
local function corner(r) return new("UICorner", { CornerRadius = UDim.new(0, r) }) end
local function pad(l, t, r, b)
    return new("UIPadding", {
        PaddingLeft = UDim.new(0, l), PaddingTop = UDim.new(0, t),
        PaddingRight = UDim.new(0, r or l), PaddingBottom = UDim.new(0, b or t),
    })
end

local function accentize(obj, prop)
    table.insert(AccentItems, { obj, prop })
    obj[prop] = Library.Accent
end

local function isPress(i)
    return i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch
end
local function isMove(i)
    return i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch
end
-- Vị trí trong hệ toạ độ ScreenGui (IgnoreGuiInset = true) -> cộng thêm inset
local function screenPos(i)
    return Vector2.new(i.Position.X, i.Position.Y) + GuiService:GetGuiInset()
end

local function ensureIridAnim()
    if iridConn then return end
    local t0 = os.clock()
    iridConn = RunService.Heartbeat:Connect(function()
        local rot = (225 + (os.clock() - t0) * 45) % 360
        for i = #Irid, 1, -1 do
            local g = Irid[i]
            if g.Parent then g.Rotation = rot else table.remove(Irid, i) end
        end
    end)
end

-- Viền kính. uniform=true: viền đều kiểu "outline" (hàng, nút); false: sáng 2 góc (cửa sổ)
local function glassStroke(parent, thickness, strength, uniform)
    local s = new("UIStroke", {
        Thickness = thickness or 1.5, Color = WHITE, Transparency = strength or 0.3,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
    local seq = uniform and NumberSequence.new({ kp(0, 0), kp(0.5, 0.4), kp(1, 0) })
        or NumberSequence.new({ kp(0, 0), kp(0.3, 0.6), kp(0.5, 0.92), kp(0.75, 0.6), kp(1, 0.05) })
    new("UIGradient", { Rotation = 45, Transparency = seq }).Parent = s
    s.Parent = parent
    return s
end

local function glassShine(parent, radius)
    local f = new("Frame", {
        Name = "Shine", BackgroundColor3 = WHITE, BackgroundTransparency = 0.7,
        Size = UDim2.fromScale(1, 1), BorderSizePixel = 0, ZIndex = parent.ZIndex, Active = false,
    }, { corner(radius) })
    new("UIGradient", {
        Rotation = 90,
        Transparency = NumberSequence.new({ kp(0, 0.1), kp(0.2, 0.55), kp(0.5, 0.93), kp(0.78, 1), kp(1, 0.6) }),
    }).Parent = f
    f.Parent = parent
    return f
end

-- o.light = true: chỉ nền + viền + bóng (nhẹ, dùng cho hàng/nút nhỏ)
-- o.themed = true: màu nền theo theme
-- o.uniform = true: viền đều
local function glass(parent, props, radius, transparency, o)
    o = o or {}
    local r = radius or 12
    local f = new("Frame", { BackgroundColor3 = WHITE, BackgroundTransparency = transparency or 0.5, BorderSizePixel = 0 })
    for k, v in pairs(props or {}) do f[k] = v end
    corner(r).Parent = f
    glassStroke(f, o.uniform and 1.3 or 1.5, o.uniform and 0.36 or 0.2, o.uniform)
    glassShine(f, r)
    if not o.light and r >= 16 then
        -- điểm chói như ánh đèn phản chiếu trên mặt kính cong
        local glint = new("Frame", {
            Name = "Glint", BackgroundColor3 = WHITE, BackgroundTransparency = 0.78, BorderSizePixel = 0,
            Position = UDim2.new(0.07, 0, 0, 5), Size = UDim2.new(0.46, 0, 0, math.max(6, math.floor(r * 0.34))),
            Active = false, ZIndex = f.ZIndex,
        }, { new("UICorner", { CornerRadius = UDim.new(1, 0) }) })
        new("UIGradient", { Transparency = NumberSequence.new({ kp(0, 0.15), kp(0.55, 0.7), kp(1, 1) }) }).Parent = glint
        glint.Parent = f
    end
    if not o.light then
        -- đáy tối nhẹ: tách lớp kính
        local depth = new("Frame", {
            Name = "Depth", BackgroundColor3 = Color3.new(0, 0, 0), BorderSizePixel = 0,
            Size = UDim2.fromScale(1, 1), Active = false,
        }, { corner(r) })
        new("UIGradient", { Rotation = 90, Transparency = NumberSequence.new({ kp(0, 1), kp(0.6, 1), kp(1, 0.82) }) }).Parent = depth
        depth.Parent = f
        -- viền khúc xạ trong, màu cầu vồng xoay
        if r >= 16 then
            local rim = new("Frame", {
                Name = "Rim", BackgroundTransparency = 1, Position = UDim2.fromOffset(3, 3),
                Size = UDim2.new(1, -6, 1, -6), Active = false,
            }, { corner(r - 3) })
            local st = new("UIStroke", { Thickness = 1.3, Color = WHITE, Transparency = 0.4, ApplyStrokeMode = Enum.ApplyStrokeMode.Border })
            local ig = new("UIGradient", {
                Rotation = 225,
                Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(140, 230, 255)),
                    ColorSequenceKeypoint.new(0.35, WHITE),
                    ColorSequenceKeypoint.new(0.65, Color3.fromRGB(255, 160, 235)),
                    ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 235, 150)),
                }),
                Transparency = NumberSequence.new({ kp(0, 0.1), kp(0.35, 0.85), kp(0.65, 0.85), kp(1, 0.1) }),
            })
            ig.Parent = st
            table.insert(Irid, ig)
            ensureIridAnim()
            st.Parent = rim
            rim.Parent = f
        end
    end
    if not o.light and r >= 20 then
        -- tán sắc: hai viền đỏ/xanh lệch nhau 1px như mép thấu kính thật
        for _, c in ipairs({ { Color3.fromRGB(255, 90, 130), 1 }, { Color3.fromRGB(80, 210, 255), -1 } }) do
            local ch = new("Frame", {
                Name = "Chroma", BackgroundTransparency = 1, Position = UDim2.fromOffset(c[2], 0),
                Size = UDim2.fromScale(1, 1), Active = false,
            }, { corner(r) })
            new("UIStroke", { Thickness = 1, Color = c[1], Transparency = 0.8, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }).Parent = ch
            ch.Parent = f
        end
    end
    f:SetAttribute("GT", f.BackgroundTransparency)
    table.insert(GlassItems, {
        f = f, base = f.BackgroundColor3, themed = o.themed,
        clear = math.min(0.98, f.BackgroundTransparency + 0.3), tinted = math.max(0.2, f.BackgroundTransparency - 0.3),
    })
    f.Parent = parent
    return f
end

local function applyGlass(animate)
    local level = Library.GlassLevel
    local alive = {}
    for _, it in ipairs(GlassItems) do
        if it.f and it.f.Parent then
            table.insert(alive, it)
            local t = it.clear + (it.tinted - it.clear) * level
            local base = it.themed and Library.Tint or it.base
            local col
            if level < 0.5 then col = base:Lerp(WHITE, (0.5 - level) * 0.6)
            else col = base:Lerp(Color3.fromRGB(24, 28, 38), (level - 0.5) * 1.4) end
            it.f:SetAttribute("GT", t)
            if animate then tween(it.f, 0.25, { BackgroundTransparency = t, BackgroundColor3 = col })
            else it.f.BackgroundTransparency = t; it.f.BackgroundColor3 = col end
        end
    end
    GlassItems = alive
    for _, w in ipairs(Library.Windows) do
        if w.Blur and w.Visible then tween(w.Blur, 0.3, { Size = 14 + level * 14 }) end
    end
end

function Library:SetGlass(level)
    Library.GlassLevel = math.clamp(level or 0.5, 0, 1)
    applyGlass(true)
end

function Library:SetEffects(on)
    Library.Effects = on and true or false
    for _, w in ipairs(Library.Windows) do
        if w.Fx and w.Fx.Parent then w.Fx.Visible = Library.Effects end
    end
end

function Library:SetTheme(name)
    local th = Library.Themes[name]
    if not th then return end
    Library.ThemeName, Library.Tint, Library.Accent = name, th.Tint, th.Accent
    local alive = {}
    for _, e in ipairs(AccentItems) do
        if e[1] and e[1].Parent then table.insert(alive, e); tween(e[1], 0.3, { [e[2]] = th.Accent }) end
    end
    AccentItems = alive
    applyGlass(true)
end

    if ok and h then return h end
    local ok2 = pcall(function() return game:GetService("CoreGui"):GetChildren() end)

end

local notifyGui, notifyHolder
function Library:Notify(title, content, duration)
    if not notifyGui or not notifyGui.Parent then
        notifyGui = new("ScreenGui", { Name = "FluentGlassNotify", ResetOnSpawn = false, IgnoreGuiInset = true, DisplayOrder = 999 })
        notifyGui.Parent = gui_parent()
        notifyHolder = new("Frame", {
            BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0),
            Position = UDim2.new(1, -12, 0, 12), Size = UDim2.fromOffset(270, 400),
        }, { new("UIListLayout", { Padding = UDim.new(0, 8), HorizontalAlignment = Enum.HorizontalAlignment.Right, SortOrder = Enum.SortOrder.LayoutOrder }) })
        notifyHolder.Parent = notifyGui
    end
    local card = glass(notifyHolder, { Size = UDim2.fromOffset(270, 0), AutomaticSize = Enum.AutomaticSize.Y }, 20, 0.4, { themed = true })
    card.Position = UDim2.fromOffset(320, 0)
    pad(14, 10).Parent = card
    local list = new("UIListLayout", { Padding = UDim.new(0, 2), SortOrder = Enum.SortOrder.LayoutOrder })
    -- UIListLayout chỉ áp cho 2 nhãn: đặt trong khung chứa riêng
    local body = new("Frame", { BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, ZIndex = 3, Parent = card }, { list })
    new("TextLabel", {
        BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 18), Font = Enum.Font.GothamBold, TextSize = 14, LayoutOrder = 1,
        TextColor3 = TEXT, TextXAlignment = Enum.TextXAlignment.Left, Text = tostring(title), ZIndex = 3, Parent = body,
    })
    new("TextLabel", {
        BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, LayoutOrder = 2,
        Font = Enum.Font.Gotham, TextSize = 13, TextWrapped = true, TextColor3 = SUB,
        TextXAlignment = Enum.TextXAlignment.Left, Text = tostring(content or ""), ZIndex = 3, Parent = body,
    })
    tween(card, 0.7, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Elastic)
    local function dismiss()
        if card.Parent then
            tween(card, 0.3, { Position = UDim2.fromOffset(320, 0) })
            task.wait(0.32)
            card:Destroy()
        end
    end
    if duration == 0 then
        local x = new("TextButton", {
            BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, 0, 0, -4), Size = UDim2.fromOffset(22, 22),
            Text = "✕", Font = Enum.Font.GothamBold, TextSize = 13, TextColor3 = SUB, ZIndex = 6, Parent = card,
        })
        x.MouseButton1Click:Connect(function() task.spawn(dismiss) end)
    else
        task.delay(duration or 4, dismiss)
    end
end

-- ========== Panel dropdown nổi ==========
local activePanel
local function closePanel()
    if activePanel then
        local p = activePanel
        activePanel = nil
        p()
    end
end

local function openPanel(screen, anchor, options, selected, onPick)
    closePanel()
    local catcher = new("TextButton", {
        BackgroundTransparency = 1, Text = "", AutoButtonColor = false, Size = UDim2.fromScale(1, 1), ZIndex = 60, Parent = screen,
    })
    local ap, as, vp = anchor.AbsolutePosition, anchor.AbsoluteSize, screen.AbsoluteSize
    local pw = math.max(as.X, 220)
    local ph = math.min(64 + #options * 40, 290)
    local x = math.clamp(ap.X + as.X - pw, 8, math.max(8, vp.X - pw - 8))
    local y = ap.Y + as.Y + 8
    if y + ph > vp.Y - 8 then y = math.max(8, ap.Y - ph - 8) end

    local panel = glass(screen, {
        Size = UDim2.fromOffset(as.X, as.Y), Position = UDim2.fromOffset(ap.X, ap.Y), ZIndex = 61, ClipsDescendants = true,
    }, 22, 0.38, { themed = true })
    applyGlass(false)

    local sbox = new("Frame", {
        BackgroundColor3 = WHITE, BackgroundTransparency = 0.9, Position = UDim2.fromOffset(10, 10), Size = UDim2.new(1, -20, 0, 34), ZIndex = 63, Parent = panel,
    }, { corner(17) })
    glassStroke(sbox, 1.2, 0.45, true)
    new("TextLabel", { BackgroundTransparency = 1, Position = UDim2.fromOffset(12, 0), Size = UDim2.fromOffset(18, 34), Text = "⌕", Font = Enum.Font.GothamBold, TextSize = 17, TextColor3 = SUB, ZIndex = 64, Parent = sbox })
    local search = new("TextBox", {
        BackgroundTransparency = 1, Position = UDim2.fromOffset(34, 0), Size = UDim2.new(1, -44, 1, 0), ClearTextOnFocus = false, Text = "",
        PlaceholderText = "Search...", PlaceholderColor3 = SUB, Font = Enum.Font.Gotham, TextSize = 13, TextColor3 = TEXT,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 64, Parent = sbox,
    })

    local list = new("ScrollingFrame", {
        BackgroundTransparency = 1, Position = UDim2.fromOffset(6, 52), Size = UDim2.new(1, -12, 1, -58), BorderSizePixel = 0,
        CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollBarThickness = 3, ScrollBarImageTransparency = 0.4, ZIndex = 63, Parent = panel,
    }, { new("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder }) })

    local function build(filter)
        for _, c in ipairs(list:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
        for idx, opt in ipairs(options) do
            local name = tostring(opt)
            if filter == "" or string.find(string.lower(name), filter, 1, true) then
                local on = (opt == selected)
                local b = new("TextButton", {
                    BackgroundColor3 = WHITE, BackgroundTransparency = on and 0.82 or 1, Size = UDim2.new(1, -4, 0, 36), LayoutOrder = idx,
                    AutoButtonColor = false, Text = name, Font = Enum.Font.Gotham, TextSize = 14, TextColor3 = TEXT,
                    TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 64, Parent = list,
                }, { corner(18), pad(16, 0, 30, 0) })
                if on then
                    glassStroke(b, 1.3, 0.35, true)
                    local bar = new("Frame", { BorderSizePixel = 0, Position = UDim2.new(0, 3, 0.5, -9), Size = UDim2.fromOffset(3, 18), ZIndex = 65, Parent = b }, { corner(2) })
                    accentize(bar, "BackgroundColor3")
                    new("TextLabel", { BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -4, 0.5, 0), Size = UDim2.fromOffset(16, 16), Text = "●", TextSize = 10, Font = Enum.Font.GothamBold, TextColor3 = TEXT, ZIndex = 65, Parent = b })
                end
                b.MouseEnter:Connect(function() if not on then tween(b, 0.15, { BackgroundTransparency = 0.9 }) end end)
                b.MouseLeave:Connect(function() if not on then tween(b, 0.2, { BackgroundTransparency = 1 }) end end)
                b.MouseButton1Click:Connect(function()
                    selected = opt
                    onPick(opt)
                    closePanel()
                end)
            end
        end
    end
    build("")
    search:GetPropertyChangedSignal("Text"):Connect(function() build(string.lower(search.Text)) end)

    tween(panel, 0.8, { Size = UDim2.fromOffset(pw, ph), Position = UDim2.fromOffset(x, y) }, Enum.EasingStyle.Elastic)

    activePanel = function()
        catcher:Destroy()
        tween(panel, 0.25, { Size = UDim2.fromOffset(as.X, as.Y), Position = UDim2.fromOffset(ap.X, ap.Y), BackgroundTransparency = 1 })
        task.delay(0.27, function() panel:Destroy() end)
    end
    catcher.MouseButton1Click:Connect(closePanel)
end

-- ========== Cửa sổ ==========
function Library:CreateWindow(cfg)
    cfg = cfg or {}
    if cfg.Theme then Library:SetTheme(cfg.Theme) end
    local Window = { Tabs = {}, Visible = true, Maximized = false }

    local screen = new("ScreenGui", { Name = "FluentGlassLiquid", ResetOnSpawn = false, IgnoreGuiInset = true, DisplayOrder = 100, ZIndexBehavior = Enum.ZIndexBehavior.Sibling })
    screen.Parent = gui_parent()
    Window.Gui = screen

    local blur
    if cfg.Blur ~= false then
        blur = new("BlurEffect", { Name = "FluentGlassBlur", Size = 0 })
        blur.Parent = Lighting
        Window.Blur = blur
    end

    local cam = workspace.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(900, 600)
    local w = math.min(cfg.Width or 660, vp.X - 24)
    local h = math.min(cfg.Height or 410, vp.Y - 48)
    local compact = w < 540
    local sideW = compact and 56 or 190
    Library.ContentWidth = w - sideW - 56
    local curW, curH = w, h

    local main = glass(screen, {
        Name = "Main", AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromOffset(w, h),
    }, 28, 0.42, { themed = true })
    applyGlass(false)
    Window.Main = main

    main.Size = UDim2.fromOffset(w * 0.9, h * 0.9)
    tween(main, 0.9, { Size = UDim2.fromOffset(w, h) }, Enum.EasingStyle.Elastic)
    if blur then tween(blur, 0.6, { Size = 14 + Library.GlassLevel * 14 }) end

    -- Vạch sáng mép trên + đèn rọi bám theo ngón tay
    local topLine = new("Frame", { BackgroundColor3 = WHITE, BorderSizePixel = 0, Active = false, ZIndex = 2, Position = UDim2.new(0, 40, 0, 2), Size = UDim2.new(1, -80, 0, 1), Parent = main })
    new("UIGradient", { Transparency = NumberSequence.new({ kp(0, 1), kp(0.5, 0.15), kp(1, 1) }) }).Parent = topLine

    local glowClip = new("Frame", { BackgroundTransparency = 1, ClipsDescendants = true, Active = false, ZIndex = 2, Position = UDim2.fromOffset(14, 14), Size = UDim2.new(1, -28, 1, -28), Parent = main })
    -- Lớp "chất lỏng": cực quang trôi chậm + bong bóng nổi (tắt bằng Library:SetEffects(false))
    local fx = new("Frame", { Name = "Fx", BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1), Active = false, ZIndex = 2, Visible = Library.Effects, Parent = glowClip })
    Window.Fx = fx
    local blobs, bubbles = {}, {}
    for k = 1, 3 do
        local root = new("Frame", { BackgroundTransparency = 1, Size = UDim2.fromOffset(0, 0), Active = false, ZIndex = 2, Parent = fx })
        for _, d in ipairs({ { 300, 0.95 }, { 190, 0.93 }, { 100, 0.91 } }) do
            local c = new("Frame", {
                AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(d[1], d[1]), BorderSizePixel = 0,
                BackgroundTransparency = d[2], Active = false, ZIndex = 2, Parent = root,
            }, { new("UICorner", { CornerRadius = UDim.new(1, 0) }) })
            accentize(c, "BackgroundColor3")
        end
        blobs[k] = root
    end
    for k = 1, 8 do
        local sz = 12 + (k * 7) % 28
        local b = new("Frame", {
            BackgroundColor3 = WHITE, BackgroundTransparency = 0.92, Size = UDim2.fromOffset(sz, sz), BorderSizePixel = 0,
            Active = false, ZIndex = 2, Parent = fx,
        }, { new("UICorner", { CornerRadius = UDim.new(1, 0) }) })
        new("UIStroke", { Thickness = 1, Color = WHITE, Transparency = 0.5, Parent = b })
        bubbles[k] = { f = b, x = (k * 0.137) % 1, y = (k * 0.31) % 1, sp = 0.02 + 0.012 * (k % 4), ph = k * 1.7 }
    end
    local fxConn = RunService.Heartbeat:Connect(function(dt)

    local function selectTab(tab)
        closePanel()
        local newIndex = table.find(Window.Tabs, tab) or 1
        local first = (currentIndex == nil)
        local dir = (currentIndex and newIndex < currentIndex) and -1 or 1
        currentIndex = newIndex
        for _, t in ipairs(Window.Tabs) do
            local on = (t == tab)
            if on and not t.Page.Visible and not first and Library.Effects then
                -- nội dung trượt vào theo chiều đổi tab
                t.Page.Position = UDim2.fromOffset(0, dir * 24)
                tween(t.Page, 0.45, { Position = UDim2.new() }, Enum.EasingStyle.Quint)
            end
            t.Page.Visible = on
            tween(t.Label, 0.25, { TextColor3 = on and TEXT or SUB })
            tween(t.Bar, 0.25, { BackgroundTransparency = on and 0 or 1 })
        end
        lensToken = lensToken + 1
        local token = lensToken
        task.defer(function()
            RunService.Heartbeat:Wait()
            if not tab.Button.Parent or not tabsArea.Parent then return end
            local newTop = tab.Button.AbsolutePosition.Y - tabsArea.AbsolutePosition.Y
            if not lens.Visible then
                lens.Position = UDim2.fromOffset(0, newTop)
                lens.Size = UDim2.new(1, 0, 0, 40)
                lens.Visible = true
                lensTop = newTop
                return
            end
            local oldTop = lensTop
            lensTop = newTop
            if oldTop == newTop then return end
            if not Library.Effects then
                lens.Position = UDim2.fromOffset(0, newTop)
                lens.Size = UDim2.new(1, 0, 0, 40)
                return
            end
            -- giọt nước: mép đi trước kéo dãn ra, mép sau bám theo rồi co lại
            lens.Position = UDim2.fromOffset(0, oldTop)
            lens.Size = UDim2.new(1, 0, 0, 40)
            local reach = math.abs(newTop - oldTop) + 40
            if dir > 0 then
                tween(lens, 0.2, { Size = UDim2.new(1, 0, 0, reach) }, Enum.EasingStyle.Quad).Completed:Connect(function()
                    if token ~= lensToken then return end
                    playRipple(newTop)
                    tween(lens, 0.55, { Position = UDim2.fromOffset(0, newTop), Size = UDim2.new(1, 0, 0, 40) }, Enum.EasingStyle.Back)
                end)
            else
                tween(lens, 0.2, { Position = UDim2.fromOffset(0, newTop), Size = UDim2.new(1, 0, 0, reach) }, Enum.EasingStyle.Quad).Completed:Connect(function()
                    if token ~= lensToken then return end
                    playRipple(newTop)
                    tween(lens, 0.55, { Size = UDim2.new(1, 0, 0, 40) }, Enum.EasingStyle.Back)
                end)
            end
        end)
    end

        local function card(height, title, desc, reserve, descLines)
            local right = (reserve or 0) + 28

            local function linesFor(text, charW)
                local n = 0
                for part in (tostring(text) .. "\n"):gmatch("(.-)\n") do
                    n = n + math.max(1, math.ceil(#part / math.max(math.floor(avail / charW), 6)))
                end
                return n
            end
            local tLines = math.min(linesFor(title, 8), 3)
            local dLines = desc and (descLines or math.min(linesFor(desc, 6.4), 6)) or 0
            local need = 18 + tLines * 18 + (desc and (4 + dLines * 15) or 0)

-- Cầu nối: giao diện FluentGlass Liquid (của bạn) -> API kiểu Fluent mà script dùng
local function RaitoCreateGlassLib()
	local Library = RaitoLoadUserGlass()
	local Adapter = { Options = {}, Windows = {}, Unloaded = false }

	local ICONS = { dices = "🎲", swords = "⚔", home = "⌂", settings = "⚙" }

	local function safeCall(fn, ...)
		if type(fn) ~= "function" then
			return
		end
		local ok, err = pcall(fn, ...)
		if not ok then
			warn("[Raito UI] " .. tostring(err))
		end
	end

	local function normalizeImage(image)
		if type(image) == "number" or (type(image) == "string" and image:match("^%d+$")) then
			return "rbxassetid://" .. tostring(image)
		end
		if type(image) == "string" and image ~= "" then
			return image
		end
		return nil
	end

	local function register(id, object)
		if id ~= nil then
			Adapter.Options[id] = object
		end
		object.Id = id
		object.Changed = {}
		function object:OnChanged(callback)
			self.Changed[#self.Changed + 1] = callback
		end
		return object
	end

	local function fireChanged(object, value)
		for _, callback in ipairs(object.Changed) do
			safeCall(callback, value)
		end
	end

	function Adapter:Notify(config)
		config = config or {}
		if Adapter.Unloaded then
			return
		end
		pcall(function()
			Library:Notify(tostring(config.Title or ""), tostring(config.Content or ""), tonumber(config.Duration) or 4)
		end)
	end

	function Adapter:Destroy()
		Adapter.Unloaded = true
		for _, window in ipairs(Adapter.Windows) do
			pcall(function()
				window:Destroy()
			end)
		end
		Adapter.Windows = {}
		pcall(function()
			local holders = {}
			if type(gethui) == "function" then
				holders[#holders + 1] = gethui()
			end
			holders[#holders + 1] = game:GetService("CoreGui")
			for _, holder in ipairs(holders) do
				local gui = holder:FindFirstChild("FluentGlassNotify")
				if gui then
					gui:Destroy()
				end
			end
		end)
	end

	local function wrapTab(rawTab)
		local Tab = {}

		function Tab:AddSection(name)
			rawTab:AddSection(tostring(name or ""))
			return {}
		end

		function Tab:AddParagraph(config)
			config = config or {}
			local content = config.Content
			if content ~= nil then
				content = tostring(content)
			end
			local label = rawTab:AddLabel({
				Title = tostring(config.Title or ""),
				Description = content,
				Lines = config.Lines,
			})
			local object = {}
			function object:SetTitle(text)
				label:Set(text)
			end
			function object:SetDesc(text)
				label:SetDesc(text)
			end
			object.SetValue = object.SetDesc
			return object
		end

		function Tab:AddButton(config)
			config = config or {}
			local description = config.Description
			if description == "" then
				description = nil
			end
			rawTab:AddButton({
				Title = tostring(config.Title or ""),
				Description = description,
				Callback = config.Callback,
			})
			return {}
		end

		function Tab:AddToggle(id, config)
			config = config or {}
			local object = register(id, { Type = "Toggle", Value = config.Default == true })
			local description = config.Description
			if description == "" then
				description = nil
			end
			local raw = rawTab:AddToggle({
				Title = tostring(config.Title or ""),
				Description = description,
				Default = object.Value,
				Callback = function(value)
					object.Value = value == true
					safeCall(config.Callback, object.Value)
					fireChanged(object, object.Value)
				end,
			})
			function object:SetValue(value, silent)
				value = value == true
				if object.Value == value then
					return
				end
				object.Value = value
				raw:Set(value, true)
				if not silent then
					safeCall(config.Callback, value)
					fireChanged(object, value)
				end
			end
			return object
		end

		function Tab:AddSlider(id, config)
			config = config or {}
			local min = tonumber(config.Min) or 0
			local max = tonumber(config.Max) or 100
			if max <= min then
				max = min + 1
			end
			local rounding = tonumber(config.Rounding) or 0
			local mult = 10 ^ rounding
			local function fit(value)
				value = tonumber(value) or min
				return math.clamp(math.floor(value * mult + 0.5) / mult, min, max)
			end
			local object = register(id, { Type = "Slider", Value = fit(config.Default or min) })
			local description = config.Description
			if description == "" then
				description = nil
			end
			local raw = rawTab:AddSlider({
				Title = tostring(config.Title or ""),
				Description = description,
				Min = min,
				Max = max,
				Default = object.Value,
				Rounding = rounding,
				Callback = function(value)
					object.Value = fit(value)
					safeCall(config.Callback, object.Value)
					fireChanged(object, object.Value)
				end,
			})
			function object:SetValue(value, silent)
				value = fit(value)
				if value == object.Value then
					return
				end
				object.Value = value
				raw:Set(value, true)
				if not silent then
					safeCall(config.Callback, value)
					fireChanged(object, value)
				end
			end
			return object
		end

		function Tab:AddDropdown(id, config)
			config = config or {}
			local values = config.Values or {}
			local default = config.Default
			if type(default) == "number" then
				default = values[default]
			end
			local object = register(id, { Type = "Dropdown", Values = values, Value = default })
			local description = config.Description
			if description == "" then
				description = nil
			end
			local raw = rawTab:AddDropdown({
				Title = tostring(config.Title or ""),
				Description = description,
				Options = values,
				Default = default,
				Callback = function(value)
					object.Value = value
					safeCall(config.Callback, value)
					fireChanged(object, value)
				end,
			})
			function object:SetValues(list)
				object.Values = list or {}
				raw:SetOptions(object.Values)
				if object.Value ~= nil and not table.find(object.Values, object.Value) then
					object.Value = nil
					raw:Set(nil, true)
				end
			end
			function object:SetValue(value, silent)
				if type(value) == "number" then
					value = object.Values[value]
				end
				object.Value = value
				raw:Set(value, true)
				if not silent then
					safeCall(config.Callback, value)
					fireChanged(object, value)
				end
			end
			return object
		end

		function Tab:AddInput(id, config)
			config = config or {}
			local object = register(id, { Type = "Input", Value = tostring(config.Default or "") })
			local description = config.Description
			if description == "" then
				description = nil
			end
			local raw = rawTab:AddInput({
				Title = tostring(config.Title or ""),
				Description = description,
				Placeholder = config.Placeholder,
				Callback = function(text)
					object.Value = tostring(text or "")
					safeCall(config.Callback, object.Value)
					fireChanged(object, object.Value)
				end,
			})
			if object.Value ~= "" then
				raw:Set(object.Value)
			end
			function object:SetValue(value, silent)
				value = tostring(value or "")
				object.Value = value
				raw:Set(value)
				if not silent then
					safeCall(config.Callback, value)
					fireChanged(object, value)
				end
			end
			return object
		end

		return Tab
	end

	function Adapter:CreateWindow(config)
		config = config or {}
		local size = config.Size
		local raw = Library:CreateWindow({
			Title = config.Title,
			SubTitle = config.SubTitle,
			Theme = config.GlassTheme or "Obsidian White",
			Width = size and size.X.Offset or nil,
			Height = size and size.Y.Offset or nil,
			OrbImage = normalizeImage(config.Image),
		})
		Adapter.Windows[#Adapter.Windows + 1] = raw

		local Window = { Raw = raw, Root = raw.Main }

		function Window:AddTab(tabConfig)
			tabConfig = tabConfig or {}
			local icon = ICONS[tabConfig.Icon] or tabConfig.Emoji
			return wrapTab(raw:AddTab(tostring(tabConfig.Title or "Tab"), icon))
		end

		function Window:SelectTab(index)
			raw:SelectTab(index)
		end

		function Window:Minimize()
			raw:Toggle()
		end

		function Window:Destroy()
			raw:Destroy()
		end

		return Window
	end

	return Adapter
end

local function saveJson(path, data)
	local ok, text = pcall(function()
		return HttpService:JSONEncode(data)
	end)
	if not ok then
		return false
	end
	memoryFiles[path] = text
	if canFiles() then
		ensureFolders()
		return pcall(writefile, path, text)
	end
	return true
end

local function loadJson(path)
	local text = memoryFiles[path]
	if canFiles() then
		local okExists, exists = pcall(function()
			return type(isfile) ~= "function" or isfile(path)
		end)
		if okExists and exists then
			local okRead, content = pcall(readfile, path)
			if okRead and type(content) == "string" then
				text = content
			end
		end
	end
	if not text then
		return nil
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(text)
	end)
	return ok and data or nil
end

local function instancePath(inst)
	local names = {}
	local current = inst
	while current and current ~= game do
		table.insert(names, 1, current.Name)
		current = current.Parent
	end
	return names
end

local function resolvePath(names)
	if type(names) ~= "table" or #names == 0 then
		return nil
	end
	local current = game
	for index, name in ipairs(names) do
		local nextObject
		if index == 1 then
			local ok, service = pcall(function()
				return game:GetService(name)
			end)
			nextObject = ok and service or game:FindFirstChild(name)
		else
			nextObject = current:FindFirstChild(name)
		end
		if not nextObject then
			return nil
		end
		current = nextObject
	end
	return current
end

local function toHex(text)
	return (text:gsub(".", function(c)
		return string.format("%02x", string.byte(c))
	end))
end

local function fromHex(hex)
	return (hex:gsub("%x%x", function(h)
		return string.char(tonumber(h, 16))
	end))
end

local function serialize(value, depth)
	depth = depth or 0
	local kind = typeof(value)
	if depth > 6 then
		return { t = "nil" }
	end
	if value == nil then
		return { t = "nil" }
	elseif kind == "number" then
		if value ~= value then
			return { t = "num", v = "nan" }
		elseif value == math.huge then
			return { t = "num", v = "inf" }
		elseif value == -math.huge then
			return { t = "num", v = "-inf" }
		end
		return { t = "n", v = value }
	elseif kind == "string" then
		if utf8.len(value) == nil or string.find(value, "%c") then
			return { t = "hs", v = toHex(value) }
		end
		return { t = "s", v = value }
	elseif kind == "boolean" then
		return { t = "b", v = value }
	elseif kind == "buffer" and type(buffer) == "table" then
		return { t = "buf", v = toHex(buffer.tostring(value)) }
	elseif kind == "Vector3" then
		return { t = "v3", v = { value.X, value.Y, value.Z } }
	elseif kind == "Vector2" then
		return { t = "v2", v = { value.X, value.Y } }
	elseif kind == "CFrame" then
		return { t = "cf", v = { value:GetComponents() } }
	elseif kind == "Color3" then
		return { t = "c3", v = { value.R, value.G, value.B } }
	elseif kind == "Instance" then
		return { t = "i", v = instancePath(value) }
	elseif kind == "table" then
		local entries = {}
		for key, item in pairs(value) do
			entries[#entries + 1] = { k = serialize(key, depth + 1), v = serialize(item, depth + 1) }
		end
		return { t = "t", v = entries }
	end
	State.SerializeWarning = kind
	return { t = "nil" }
end

local function deserialize(node)
	if type(node) ~= "table" then
		return nil
	end
	local tag, v = node.t, node.v
	if tag == "n" or tag == "s" or tag == "b" then
		return v
	elseif tag == "hs" then
		return fromHex(v)
	elseif tag == "buf" then
		if type(buffer) == "table" then
			return buffer.fromstring(fromHex(v))
		end
		return nil
	elseif tag == "num" then
		if v == "inf" then
			return math.huge
		elseif v == "-inf" then
			return -math.huge
		end
		return 0 / 0
	elseif tag == "v3" then
		return Vector3.new(v[1], v[2], v[3])
	elseif tag == "v2" then
		return Vector2.new(v[1], v[2])
	elseif tag == "cf" then
		return CFrame.new(table.unpack(v))
	elseif tag == "c3" then
		return Color3.new(v[1], v[2], v[3])
	elseif tag == "i" then
		return resolvePath(v)
	elseif tag == "t" then
		local result = {}
		for _, entry in ipairs(v) do
			local key = deserialize(entry.k)
			if key ~= nil then
				result[key] = deserialize(entry.v)
			end
		end
		return result
	end
	return nil
end

local function serializeArgs(packed)
	local list = {}
	for index = 1, packed.n do
		list[index] = serialize(packed[index])
	end
	return { n = packed.n, list = list }
end

local function deserializeArgs(stored)
	local args = { n = stored.n or #stored.list }
	for index = 1, args.n do
		args[index] = deserialize(stored.list[index])
	end
	return args
end

local function describeArgs(packed)
	local parts = {}
	for index = 1, math.min(packed.n, 3) do
		local value = packed[index]
		local kind = typeof(value)
		if kind == "string" or kind == "number" or kind == "boolean" then
			parts[#parts + 1] = tostring(value)
		else
			parts[#parts + 1] = kind
		end
	end
	return table.concat(parts, ", ")
end

local function isNoise(name)
	local lowered = string.lower(name)
	for _, word in ipairs(NOISE_WORDS) do
		if string.find(lowered, word, 1, true) then
			return true
		end
	end
	return false
end

local REMOTE_CLASSES = {
	RemoteEvent = true,
	RemoteFunction = true,
	UnreliableRemoteEvent = true,
}

local function installHook()
	if State.HookOk then
		return true
	end
	if type(hookmetamethod) ~= "function" or type(getnamecallmethod) ~= "function" then
		return false
	end
	local oldNamecall
	local function handler(self, ...)
		local method = getnamecallmethod()
		if (method == "FireServer" or method == "InvokeServer")
			and not (type(checkcaller) == "function" and checkcaller())
			and (State.Recording or State.LearnTarget) then
			local classOk = false
			pcall(function()
				classOk = REMOTE_CLASSES[self.ClassName] == true
			end)
			if classOk then
				-- chỉ đẩy vào hàng đợi, xử lý sau để không làm hỏng namecall
				State.Queue[#State.Queue + 1] = {
					remote = self,
					method = method,
					args = table.pack(...),
					time = os.clock(),
				}
			end
		end
		return oldNamecall(self, ...)
	end
	if type(newcclosure) == "function" then
		handler = newcclosure(handler)
	end
	local ok = pcall(function()
		oldNamecall = hookmetamethod(game, "__namecall", handler)
	end)
	State.HookOk = ok and oldNamecall ~= nil
	return State.HookOk
end

local rollParagraph, recordParagraph

local function rollStatusText()
	local function status(key)
		return State.Learned[key] and "đã học" or "chưa học"
	end
	return "Roll 1: " .. status("roll1") .. " | Roll 10: " .. status("roll10")

local function refreshRollParagraph()
	if rollParagraph then
		rollParagraph:SetDesc(rollStatusText())
	end
end

local function refreshRecordParagraph()
	if recordParagraph then
		local text = "Đã ghi " .. tostring(#State.RecordedActions) .. " lệnh"
		if State.Recording then
			text = "ĐANG GHI... " .. text
		end
		recordParagraph:SetDesc(text .. " | Lệnh gần nhất: " .. State.LastRecordedLabel)
	end
end

local function processQueue()
	if #State.Queue == 0 then
		return
	end
	local queue = State.Queue
	State.Queue = {}
	for _, entry in ipairs(queue) do
		local okName, name = pcall(function()
			return entry.remote.Name
		end)
		name = okName and name or "?"
		if not isNoise(name) then
			if State.LearnTarget then
				local key = State.LearnTarget
				State.LearnTarget = nil
				State.Learned[key] = {
					name = name,
					method = entry.method,
					path = instancePath(entry.remote),
					args = serializeArgs(entry.args),
				}
				saveJson(FOLDER .. "/learned.json", State.Learned)
				refreshRollParagraph()
				notify("Đã học lệnh", (key == "roll1" and "Roll 1" or "Roll 10") .. " = " .. name .. "(" .. describeArgs(entry.args) .. ")", 5)
			elseif State.Recording then
				State.SerializeWarning = nil
				State.RecordedActions[#State.RecordedActions + 1] = {
					t = entry.time - State.RecordStart,
					name = name,
					method = entry.method,
					path = instancePath(entry.remote),
					args = serializeArgs(entry.args),
				}
				State.LastRecordedLabel = name .. "(" .. describeArgs(entry.args) .. ")"
				if State.SerializeWarning then
					State.LastRecordedLabel = State.LastRecordedLabel .. " [cảnh báo: có đối số kiểu " .. tostring(State.SerializeWarning) .. " không lưu được]"
				end
				refreshRecordParagraph()
			end
		end
	end
end

task.spawn(function()
	while State.Running do
		pcall(processQueue)
		task.wait(0.05)
	end
end)

-- trả về: ok, kết quả
local function fireStored(stored)
	local remote = resolvePath(stored.path)
	if not remote then
		return false, "không tìm thấy remote"
	end
	local args = deserializeArgs(stored.args)
	local method = stored.method
	local ok, result = pcall(function()
		if method == "InvokeServer" then
			return remote:InvokeServer(table.unpack(args, 1, args.n))
		end
		remote:FireServer(table.unpack(args, 1, args.n))
		return true
	end)
	return ok, result
end

local function playerGui()
	return LocalPlayer:FindFirstChild("PlayerGui")
end

local function isGuiVisible(object)
	local current = object
	while current and current ~= game do
		if current:IsA("GuiObject") and not current.Visible then
			return false
		end
		if current:IsA("ScreenGui") and not current.Enabled then
			return false
		end
		current = current.Parent
	end
	return true
end

local function buttonText(button)
	if button:IsA("TextButton") and button.Text ~= "" then
		return button.Text
	end
	for _, child in ipairs(button:GetDescendants()) do
		if child:IsA("TextLabel") and child.Text ~= "" then
			return child.Text
		end
	end
	return ""
end

local function cleanText(text)
	text = string.lower(tostring(text or ""))
	text = text:gsub("<[^>]+>", "")
	text = text:gsub("^%s+", ""):gsub("%s+$", "")
	return text
end

local function findButton(words, exact)
	local gui = playerGui()
	if not gui then
		return nil
	end
	for _, object in ipairs(gui:GetDescendants()) do
		if (object:IsA("TextButton") or object:IsA("ImageButton")) and isGuiVisible(object) then
			local text = cleanText(buttonText(object))
			if text ~= "" then
				for _, word in ipairs(words) do
					local target = cleanText(word)
					if (exact and text == target) or (not exact and string.find(text, target, 1, true)) then
						return object
					end
				end
			end
		end
	end
	return nil
end

local function clickButton(button)
	if not button then
		return false
	end
	local fired = false
	if type(getconnections) == "function" then
		for _, signalName in ipairs({ "Activated", "MouseButton1Click", "MouseButton1Down", "MouseButton1Up" }) do
			pcall(function()
				for _, connection in ipairs(getconnections(button[signalName])) do
					connection:Fire()
					fired = true
				end
			end)
		end
	end
	if not fired then
		pcall(function()
			local vim = game:GetService("VirtualInputManager")
			local inset = GuiService:GetGuiInset()
			local x = button.AbsolutePosition.X + button.AbsoluteSize.X / 2
			local y = button.AbsolutePosition.Y + button.AbsoluteSize.Y / 2 + inset.Y
			vim:SendMouseButtonEvent(x, y, 0, true, game, 0)
			task.wait(0.05)
			vim:SendMouseButtonEvent(x, y, 0, false, game, 0)
			fired = true
		end)
	end
	return fired
end

local function hasMoneyWarning()
	local gui = playerGui()
	if not gui then
		return false
	end
	for _, object in ipairs(gui:GetDescendants()) do
		if object:IsA("TextLabel") and object.Text ~= "" and isGuiVisible(object) then
			local text = cleanText(object.Text)
			for _, word in ipairs(MONEY_STOP_WORDS) do
				if string.find(text, word, 1, true) then
					return true
				end
			end
		end
	end
	return false
end

local function findMoneyValues()
	local list, seen = {}, {}
	for _, object in ipairs(LocalPlayer:GetDescendants()) do
		if (object:IsA("NumberValue") or object:IsA("IntValue")) and not seen[object] then
			local name = string.lower(object.Name)
			if string.find(name, "coin", 1, true) or string.find(name, "cash", 1, true)
				or string.find(name, "gem", 1, true) or string.find(name, "money", 1, true)
				or string.find(name, "gold", 1, true) then
				seen[object] = true
				list[#list + 1] = object
			end
		end
	end
	return list
end

local NO_MONEY_ROLL_CAP = 100

local rolling = false

local function rollLoop()
	if rolling then
		return
	end
	rolling = true
	local key = Options.RollChoice.Value == "Roll 10" and "roll10" or "roll1"
	local stored = State.Learned[key]
	if not stored then
		rollStatus:SetDesc("Chưa học lệnh " .. (key == "roll10" and "Roll 10" or "Roll 1") .. ". Bấm \"Học lệnh\" trước.")
		notify("Chưa học lệnh", "Hãy học lệnh roll trước khi bấm Start Roll", 5)
		rolling = false
		if Options.StartRoll then
			Options.StartRoll:SetValue(false, true)
		end
		return
	end

	local candidates = findMoneyValues()
	local count, fails, unchanged = 0, 0, 0
	local spentEver = false
	local reason = "đã dừng"
	local hardCap = 0
	if #candidates == 0 then
		hardCap = NO_MONEY_ROLL_CAP
		notify("Không thấy tiền", "Không dò được số tiền của bạn nên script tự giới hạn " .. tostring(hardCap) .. " lần roll", 5)
	end

	while State.Running and Options.StartRoll and Options.StartRoll.Value do
		local before = {}
		for index, value in ipairs(candidates) do
			before[index] = value.Value
		end
		local ok, result = fireStored(stored)
		count = count + 1

		if not ok or result == false or result == nil and stored.method == "InvokeServer" then
			fails = fails + 1
		else
			fails = 0
		end

		task.wait(math.max(tonumber(Options.RollDelay.Value) or 0.4, 0.05) + 0.25)

		local spent = false
		local moneyText = ""
		for index, value in ipairs(candidates) do
			if value.Value < before[index] then
				spent = true
			end
			moneyText = moneyText .. (index > 1 and ", " or "") .. value.Name .. " " .. tostring(value.Value)
		end
		if #candidates > 0 then
			if spent then
				spentEver = true
				unchanged = 0
			else
				unchanged = unchanged + 1
			end
		end

		rollStatus:SetDesc("Đã roll " .. tostring(count) .. " lần" .. (moneyText ~= "" and (" | " .. moneyText) or ""))

		local limit = tonumber(Options.RollLimit.Value) or 0
		if hardCap > 0 and (limit == 0 or limit > hardCap) then
			limit = hardCap
		end
		if limit > 0 and count >= limit then
			reason = "đạt giới hạn " .. tostring(limit) .. " lần"
			break
		end
		if fails >= 3 then
			reason = "server từ chối 3 lần liên tiếp (hết tiền?)"
			break
		end
		if spentEver and unchanged >= 4 then
			reason = "tiền không giảm sau 4 lần roll (hết tiền)"
			break
		end
		if #candidates > 0 and not spentEver and count >= 10 and stored.method ~= "InvokeServer" then
			reason = "10 lần roll mà tiền không giảm: có thể học nhầm lệnh, hãy học lại"
			break
		end
		if count % 3 == 0 and hasMoneyWarning() then
			reason = "game báo không đủ tiền"
			break
		end
	end

	rollStatus:SetDesc("Dừng: " .. reason .. " | Đã roll " .. tostring(count) .. " lần")
	rolling = false
	if Options.StartRoll and Options.StartRoll.Value then
		Options.StartRoll:SetValue(false, true)
	end
end

local function playMacro(macro)
	State.Playing = true
	State.PlayToken = State.PlayToken + 1
	local token = State.PlayToken
	local actions = macro.actions or {}
	local last = 0
	for index, action in ipairs(actions) do
		if not State.Running or token ~= State.PlayToken or not (Options.PlayMacro and Options.PlayMacro.Value) then
			break
		end
		local speed = math.max(tonumber(Options.MacroSpeed.Value) or 1, 0.1)
		local wait_ = math.max((action.t - last) / speed, 0)
		last = action.t
		if wait_ > 0 then
			task.wait(wait_)
		end
		if token ~= State.PlayToken or not (Options.PlayMacro and Options.PlayMacro.Value) then
			break
		end
		local ok, result = fireStored(action)
		-- InvokeServer trả về false khi chưa đủ tiền -> thử lại vài lần
		local tries = 0
		while State.Running and token == State.PlayToken and action.method == "InvokeServer"
			and (not ok or result == false) and tries < 12 do
			tries = tries + 1
			task.wait(0.5)
			ok, result = fireStored(action)
		end
		playStatus:SetDesc("Đang chạy " .. tostring(macro.name or "?") .. ": " .. tostring(index) .. "/" .. tostring(#actions) .. " lệnh")
	end
	State.Playing = false
	State.MacroDone = true
	playStatus:SetDesc("Đã chạy xong " .. tostring(macro.name or "?"))
end

local function loadSelectedMacro()
	local name = Options.MacroSelect and Options.MacroSelect.Value
	if not name or name == "" then
		return nil
	end
	local data = loadJson(macroFilePath(name))
	if type(data) == "table" and type(data.actions) == "table" then
		data.name = data.name or name
		return data
	end
	return nil
end

local joinTween
local function flyTo(cf)
	local char = LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then
		return
	end
	local distance = (root.Position - cf.Position).Magnitude
	if distance < 6 then
		return
	end
	if joinTween then
		pcall(function()
			joinTween:Cancel()
		end)
	end
	joinTween = TweenService:Create(root, TweenInfo.new(distance / 120, Enum.EasingStyle.Linear), { CFrame = cf })
	joinTween:Play()
end

			elseif entry.placeId ~= game.PlaceId then
				joinStatus:SetDesc("Không ở lobby của map này (đang trong trận?)")
			else
				local cf = deserialize(entry.cf)
				if cf then
					joinStatus:SetDesc("Đang bay tới " .. tostring(name))
					flyTo(cf)
					task.wait(1.2)
					local button = findButton(PLAY_WORDS, true)

					end
				end
			end
		end

	end
